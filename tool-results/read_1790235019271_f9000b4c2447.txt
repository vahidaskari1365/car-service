   184→   184→  function onNodePointerDown(node: FlowNode, ev: React.PointerEvent) {
   185→   185→    if (runningRef.current) return;
   186→   186→    ev.stopPropagation();
   187→   187→    const startX = ev.clientX, startY = ev.clientY;
   188→   188→    const origX = node.x, origY = node.y;
   189→   189→    let moved = false;
   190→   190→    const onMove = (e2: PointerEvent) => {
   191→   191→      const dx = e2.clientX - startX, dy = e2.clientY - startY;
   192→   192→      if (!moved && Math.abs(dx) < 4 && Math.abs(dy) < 4) return;
   193→   193→      moved = true;
   194→   194→      setNodes(ns => ns.map(n => n.id === node.id
   195→   195→        ? { ...n, x: Math.max(4, Math.min(CANVAS_W - NODE_W - 4, origX + dx)), y: Math.max(4, Math.min(CANVAS_H - NODE_H - 4, origY + dy)) }
   196→   196→        : n));
   197→   197→    };
   198→   198→    const onUp = () => {
   199→   199→      window.removeEventListener('pointermove', onMove);
   200→   200→      window.removeEventListener('pointerup', onUp);
   201→   201→      if (moved) { touch(); return; }
   202→   202→      if (connectingRef.current) { completeConnect(node.id); return; }
   203→   203→      setSelectedNode(node.id);
   204→   204→      setSelectedEdge(null);
   205→   205→    };
   206→   206→    window.addEventListener('pointermove', onMove);
   207→   207→    window.addEventListener('pointerup', onUp);
   208→   208→  }

   293→   293→      });
   294→   294→      const json = await res.json() as RunResponse;
   295→   295→      if (!res.ok || json.error) {
   296→   296→        toast({ title: json.error || 'خطا در اجرای تست فرآیند', variant: 'destructive' });

   299→   299→      for (const r of json.results) {
   300→   300→        setActiveEdge(edges.find(e => e.to === r.nodeId)?.id || null);
   301→   301→        setStates(st => ({ ...st, [r.nodeId]: 'running' }));
   302→   302→        await sleep(640);
   303→   303→        setStates(st => ({ ...st, [r.nodeId]: r.status }));
   304→   304→        setMsgs(m => ({ ...m, [r.nodeId]: r.message }));
   305→   305→        setLog(l => [...l, { nodeId: r.nodeId, label: labelOf(r.nodeId), status: r.status, message: r.message, durationMs: r.durationMs }]);
   306→   306→        await sleep(r.status === 'error' ? 460 : 200);
   307→   307→        setActiveEdge(null);
   308→   308→      }
   309→   309→      for (const s of json.skipped) {
   310→   310→        setStates(st => ({ ...st, [s.nodeId]: 'skipped' }));
   311→   311→        setMsgs(m => ({ ...m, [s.nodeId]: s.reason }));
   312→   312→        setLog(l => [...l, { nodeId: s.nodeId, label: labelOf(s.nodeId), status: 'skipped', message: s.reason, durationMs: 0 }]);
   313→   313→      }
   314→   314→      for (const w of json.warnings) {
   315→   315→        setLog(l => [...l, { nodeId: null, label: 'هشدار ساختار', status: 'warning', message: w, durationMs: 0 }]);
   316→   316→      }
   317→   317→      setSummary(json.summary);
   318→   318→      const allOk = json.summary.failed === 0;
   319→   319→      if (flowId) {
   320→   320→        fetch('/api/flows', {
   321→   321→          method: 'PUT',
   322→   322→          headers: { 'Content-Type': 'application/json', 'X-User': encodeURIComponent(getActorName()) },
   323→   323→          body: JSON.stringify({ id: flowId, lastTestAt: new Date().toISOString(), lastTestOk: allOk }),
   324→   324→        }).catch(() => {});
   325→   325→      }
   326→   326→      const failMsg = json.results.find(r => r.status === 'error')?.message;
   327→   327→      if (allOk) {
   328→   328→        toast({ title: `تست موفق — ${faNumber(json.summary.ok)} مرحله سبز شد ✓`, description: `${flowName} · ${faNumber(json.summary.durationMs)}ms` });
   329→   329→      } else {
   330→   330→        toast({ title: 'تست با خطا متوقف شد', description: failMsg, variant: 'destructive' });
   331→   331→      }
   332→   332→    } catch {
   333→   333→      toast({ title: 'ارتباط با سرور برقرار نشد', variant: 'destructive' });
   334→   334→    } finally {
   335→   335→      setRunning(false);
   336→   336→      setActiveEdge(null);
   337→   337→    }
   338→   338→  }

   341→   341→  function edgeGeom(e: FlowEdge) {
   342→   342→    const from = nodeById(e.from), to = nodeById(e.to);
   343→   343→    if (!from || !to) return null;
   344→   344→    const x1 = from.x + NODE_W, y1 = from.y + NODE_H / 2;
   345→   345→    const x2 = to.x, y2 = to.y + NODE_H / 2;
   346→   346→    const dx = Math.max(46, Math.abs(x2 - x1) / 2);
   347→   347→    return { x1, y1, x2, y2, d: `M ${x1} ${y1} C ${x1 + dx} ${y1}, ${x2 - dx} ${y2}, ${x2} ${y2}`, mx: (x1 + x2) / 2, my: (y1 + y2) / 2 };
   348→   348→  }

   350→   350→  function edgeClass(e: FlowEdge): string {

   352→   352→    if (activeEdge === e.id) return 'stroke-amber-500 wf-edge-active';

   354→   354→    if (targetState === 'error') return 'stroke-red-500';
   355→   355→    if (targetState === 'skipped') return 'stroke-zinc-300 dark:stroke-zinc-700';
   356→   356→    return 'stroke-zinc-400 dark:stroke-zinc-600';
