   163→  function completeConnect(targetId: string) {
   164→    const from = connectingRef.current;
   165→    setConnecting(null);
   166→    setGhost(null);
   167→    if (!from || from === targetId || runningRef.current) return;
   168→    setEdges(es => {
   169→      if (es.some(e => e.from === from && e.to === targetId)) return es;
   170→      const fromNode = nodeById(from);
   171→      let label: string | undefined;
   172→      if (fromNode?.kind === 'condition') {
   173→        const outs = es.filter(e => e.from === from);
   174→        if (!outs.some(x => x.label === 'بله')) label = 'بله';
   175→        else if (!outs.some(x => x.label === 'خیر')) label = 'خیر';
   176→      }
   177→      return [...es, { id: uid('e'), from, to: targetId, label }];
   178→    });
   179→    touch();
   180→  }
   181→  function cancelConnect() { setConnecting(null); setGhost(null); }

   183→  // ─── درگ نود ───
   184→  function onNodePointerDown(node: FlowNode, ev: React.PointerEvent) {
   185→    if (runningRef.current) return;
   186→    ev.stopPropagation();
   187→    const startX = ev.clientX, startY = ev.clientY;
   188→    const origX = node.x, origY = node.y;
   189→    let moved = false;
   190→    const onMove = (e2: PointerEvent) => {
   191→      const dx = e2.clientX - startX, dy = e2.clientY - startY;
   192→      if (!moved && Math.abs(dx) < 4 && Math.abs(dy) < 4) return;
   193→      moved = true;
   194→      setNodes(ns => ns.map(n => n.id === node.id
   195→        ? { ...n, x: Math.max(4, Math.min(CANVAS_W - NODE_W - 4, origX + dx)), y: Math.max(4, Math.min(CANVAS_H - NODE_H - 4, origY + dy)) }
   196→        : n));
   197→    };
   198→    const onUp = () => {
   199→      window.removeEventListener('pointermove', onMove);
   200→      window.removeEventListener('pointerup', onUp);
   201→      if (moved) { touch(); return; }
   202→      if (connectingRef.current) { completeConnect(node.id); return; }
   203→      setSelectedNode(node.id);
   204→      setSelectedEdge(null);
   205→    };
   206→    window.addEventListener('pointermove', onMove);
   207→    window.addEventListener('pointerup', onUp);
   208→  }

   235→      } else {

   238→        if (!res.ok) throw new Error();
   239→        if (json.data?.id) setFlowId(json.data.id);

   241→      await refreshFlows();
   242→      setDirty(false);
   243→      toast({ title: 'فرآیند ذخیره شد ✓', description: flowName });
   244→    } catch {
   245→      toast({ title: 'خطا در ذخیره فرآیند', variant: 'destructive' });

   283→  async function runTest() {
   284→    if (runningRef.current || nodes.length === 0) return;
   285→    setRunning(true);
   286→    setStates(Object.fromEntries(nodes.map(n => [n.id, 'idle' as FlowNodeState])));
   287→    setMsgs({}); setLog([]); setSummary(null); setActiveEdge(null);
   288→    try {
   289→      const res = await fetch('/api/workflow/run', {
   290→        method: 'POST',
   291→        headers: { 'Content-Type': 'application/json', 'X-User': encodeURIComponent(getActorName()) },
   292→        body: JSON.stringify({ flow: { id: flowId || 'tmp', name: flowName || 'فرآیند بدون نام', nodes, edges } }),
   293→      });
   294→      const json = await res.json() as RunResponse;
   295→      if (!res.ok || json.error) {
   296→        toast({ title: json.error || 'خطا در اجرای تست فرآیند', variant: 'destructive' });
   297→        return;
   298→      }
   299→      for (const r of json.results) {
   300→        setActiveEdge(edges.find(e => e.to === r.nodeId)?.id || null);
   301→        setStates(st => ({ ...st, [r.nodeId]: 'running' }));
   302→        await sleep(640);
   303→        setStates(st => ({ ...st, [r.nodeId]: r.status }));
   304→        setMsgs(m => ({ ...m, [r.nodeId]: r.message }));
   305→        setLog(l => [...l, { nodeId: r.nodeId, label: labelOf(r.nodeId), status: r.status, message: r.message, durationMs: r.durationMs }]);
   306→        await sleep(r.status === 'error' ? 460 : 200);
   307→        setActiveEdge(null);
   308→      }
   309→      for (const s of json.skipped) {
   310→        setStates(st => ({ ...st, [s.nodeId]: 'skipped' }));
   311→        setMsgs(m => ({ ...m, [s.nodeId]: s.reason }));
   312→        setLog(l => [...l, { nodeId: s.nodeId, label: labelOf(s.nodeId), status: 'skipped', message: s.reason, durationMs: 0 }]);
   313→      }
   314→      for (const w of json.warnings) {
   315→        setLog(l => [...l, { nodeId: null, label: 'هشدار ساختار', status: 'warning', message: w, durationMs: 0 }]);
   316→      }
   317→      setSummary(json.summary);
   318→      const allOk = json.summary.failed === 0;
   319→      if (flowId) {
   320→        fetch('/api/flows', {
   321→          method: 'PUT',
   322→          headers: { 'Content-Type': 'application/json', 'X-User': encodeURIComponent(getActorName()) },
   323→          body: JSON.stringify({ id: flowId, lastTestAt: new Date().toISOString(), lastTestOk: allOk }),
   324→        }).catch(() => {});
   325→      }
   326→      const failMsg = json.results.find(r => r.status === 'error')?.message;
   327→      if (allOk) {
   328→        toast({ title: `تست موفق — ${faNumber(json.summary.ok)} مرحله سبز شد ✓`, description: `${flowName} · ${faNumber(json.summary.durationMs)}ms` });
   329→      } else {
   330→        toast({ title: 'تست با خطا متوقف شد', description: failMsg, variant: 'destructive' });
   331→      }
   332→    } catch {
   333→      toast({ title: 'ارتباط با سرور برقرار نشد', variant: 'destructive' });
   334→    } finally {
   335→      setRunning(false);
   336→      setActiveEdge(null);
   337→    }
   338→  }

   341→  function edgeGeom(e: FlowEdge) {
   342→    const from = nodeById(e.from), to = nodeById(e.to);
   343→    if (!from || !to) return null;
   344→    const x1 = from.x + NODE_W, y1 = from.y + NODE_H / 2;
   345→    const x2 = to.x, y2 = to.y + NODE_H / 2;
   346→    const dx = Math.max(46, Math.abs(x2 - x1) / 2);
   347→    return { x1, y1, x2, y2, d: `M ${x1} ${y1} C ${x1 + dx} ${y1}, ${x2 - dx} ${y2}, ${x2} ${y2}`, mx: (x1 + x2) / 2, my: (y1 + y2) / 2 };
   348→  }

   712→              success: <CheckCircle2 className="h-3.5 w-3.5 text-emerald-500 shrink-0" />,
   713→              error: <XCircle className="h-3.5 w-3.5 text-red-500 shrink-0" />,
   714→              skipped: <SkipForward className="h-3.5 w-3.5 text-zinc-400 shrink-0" />,
   715→              warning: <TriangleAlert className="h-3.5 w-3.5 text-amber-500 shrink-0" />,
   716→            };
   717→            return (
   718→              <div key={i} className={`wf-log-enter flex items-center gap-2 rounded-lg px-2 py-1.5 text-[11px] border ${l.status === 'error' ? 'bg-red-50 border-red-200 dark:bg-red-500/10 dark:border-red-500/30' : l.status === 'success' ? 'bg-emerald-50/60 border-emerald-100 dark:bg-emerald-500/5 dark:border-emerald-500/20' : 'bg-muted/40 border-transparent'}`}>
   719→                <span className="font-mono text-[9px] text-muted-foreground w-5 text-center">{faNumber(i + 1)}</span>
   720→                {iconMap[l.status]}
   721→                <b className="shrink-0 max-w-36 truncate" title={l.label}>{l.label}</b>
   722→                <span className={`truncate flex-1 ${l.status === 'error' ? 'text-red-700 dark:text-red-300' : 'text-muted-foreground'}`} title={l.message}>{l.message}</span>
   723→                {l.durationMs > 0 && <span className="shrink-0 font-mono text-[9px] text-muted-foreground">{faNumber(l.durationMs)}ms</span>}
   724→              </div>
   725→            );
   726→          })}
   727→        </div>
   728→      </div>
   729→    </div>
   730→  );
   731→}
   732→
   733→// ═══════════════════ فرآیندهای جاری (اجرای واقعی) ═══════════════════
   734→const stageTone: Record<string, string> = {
   735→  done: 'bg-emerald-50 text-emerald-700 border-emerald-200 dark:bg-emerald-500/10 dark:text-emerald-300 dark:border-emerald-500/30',
   736→  in_progress: 'bg-amber-50 text-amber-700 border-amber-200 dark:bg-amber-500/10 dark:text-amber-300 dark:border-amber-500/30',
   737→  pending: 'bg-zinc-50 text-zinc-500 border-zinc-200 dark:bg-zinc-500/10 dark:text-zinc-400 dark:border-zinc-500/30',
   738→  rejected: 'bg-red-50 text-red-600 border-red-200 dark:bg-red-500/10 dark:text-red-300 dark:border-red-500/30',
   739→};
   740→const typeLabels: Record<string, string> = {
   741→  purchase: 'خرید', repair: 'تعمیر', sale: 'فروش', rental: 'اجاره', custom: 'سفارشی',
   742→};
   743→
   744→function RunningProcesses() {
   745→  const { items: processes, loading, create, update } = useEntity<WorkflowProcess>('processes');
   746→  const { toast } = useToast();
   747→  const [open, setOpen] = useState(false);
   748→  const [detailId, setDetailId] = useState<string | null>(null);
   749→  const [form, setForm] = useState({ title: '', type: 'custom', responsible: '' });
   750→  const [logNote, setLogNote] = useState('');
   751→
   752→  const detail = processes.find(p => p.id === detailId) || null;
   753→
   754→  const stats = {
   755→    total: processes.length,
   756→    active: processes.filter(p => p.currentStageIndex < p.stages.length - 1).length,
   757→    stuck: processes.filter(p => p.stages[p.currentStageIndex]?.status === 'in_progress' && p.stages.filter(s => s.status === 'done').length > 0).length,
   758→  };
   759→
   760→  async function handleAdd() {
   761→    if (!form.title || !form.responsible) {
   762→      toast({ title: 'عنوان فرآیند و مسئول اولین مرحله الزامی است', variant: 'destructive' });
   763→      return;
   764→    }
   765→    await create({
   766→      code: `WF-${6000 + processes.length + 1}`,
   767→      type: form.type as WorkflowProcess['type'], title: form.title,
   768→      currentStageIndex: 0, createdBy: 'کاربر سیستم',
   769→      createdAt: new Date().toISOString(),
   770→      stages: [
   771→        { name: 'شروع فرآیند', responsible: form.responsible, status: 'in_progress', startedAt: new Date().toISOString(), logs: [{ at: new Date().toISOString(), by: 'کاربر سیستم', action: 'فرآیند ایجاد شد' }] },
   772→        { name: 'بررسی و تأیید', responsible: 'مدیرعامل', status: 'pending', logs: [] },
   773→        { name: 'اجرا', responsible: form.responsible, status: 'pending', logs: [] },
   774→        { name: 'کنترل و بستن', responsible: 'مدیرعامل', status: 'pending', logs: [] },
   775→      ],
   776→    } as Partial<WorkflowProcess>);
   777→    setForm({ title: '', type: 'custom', responsible: '' });
   778→    setOpen(false);
   779→    toast({ title: 'فرآیند جدید ایجاد شد' });
   780→  }
   781→
   782→  async function advance(p: WorkflowProcess) {
   783→    const stages = [...p.stages];
   784→    stages[p.currentStageIndex] = {
   785→      ...stages[p.currentStageIndex], status: 'done', completedAt: new Date().toISOString(),
   786→      logs: [...stages[p.currentStageIndex].logs, { at: new Date().toISOString(), by: 'کاربر سیستم', action: 'مرحله تکمیل شد' }],
   787→    };
   788→    const nextIdx = p.currentStageIndex + 1;
   789→    if (nextIdx < stages.length) {
   790→      stages[nextIdx] = { ...stages[nextIdx], status: 'in_progress', startedAt: new Date().toISOString() };
   791→    }
   792→    await update({ id: p.id, stages, currentStageIndex: Math.min(nextIdx, stages.length - 1) });
   793→    if (nextIdx >= stages.length - 1) {
   794→      await update({ id: p.id, currentStageIndex: stages.length - 1 });
   795→      stages[stages.length - 1] = { ...stages[stages.length - 1], status: 'done', completedAt: new Date().toISOString() };
   796→      await update({ id: p.id, stages });
   797→      toast({ title: 'فرآیند کامل شد ✓', description: p.title });
   798→    } else {
   799→      toast({ title: 'مرحله پیشرفت کرد', description: `${p.title} → ${stages[nextIdx].name}` });
   800→    }
   801→  }
   802→
   803→  async function rejectStage(p: WorkflowProcess) {
   804→    const note = window.prompt('دلیل رد مرحله:') || '';
   805→    const stages = [...p.stages];
   806→    stages[p.currentStageIndex] = {
   807→      ...stages[p.currentStageIndex], status: 'rejected',
   808→      logs: [...stages[p.currentStageIndex].logs, { at: new Date().toISOString(), by: 'کاربر سیستم', action: 'مرحله رد شد', note }],
   809→    };
   810→    await update({ id: p.id, stages });
   811→    toast({ title: 'مرحله رد شد', variant: 'destructive' });
   812→  }
   813→
   814→  async function addLog(p: WorkflowProcess) {
   815→    if (!logNote.trim()) return;
   816→    const stages = [...p.stages];
   817→    stages[p.currentStageIndex] = {
   818→      ...stages[p.currentStageIndex],
   819→      logs: [...stages[p.currentStageIndex].logs, { at: new Date().toISOString(), by: 'کاربر سیستم', action: 'یادداشت', note: logNote }],
   820→    };
   821→    await update({ id: p.id, stages });
   822→    setLogNote('');
   823→    toast({ title: 'یادداشت در لاگ فرآیند ثبت شد' });
   824→  }
   825→
   826→  return (
   827→    <div className="space-y-4">
   828→      <div className="grid grid-cols-2 lg:grid-cols-3 gap-3">
   829→        <KpiCard title="کل فرآیندها" value={faNumber(stats.total)} icon={GitBranch} tone="amber" />
   830→        <KpiCard title="فرآیندهای جاری" value={faNumber(stats.active)} icon={ArrowLeft} tone="teal" />
   831→        <KpiCard title="در جریان اجرا" value={faNumber(stats.stuck)} sub="مرحله فعال" icon={History} tone="zinc" />
   832→      </div>
   833→
   834→      {loading ? <LoadingTable rows={4} /> : (
   835→        <div className="space-y-4">
   836→          {processes.map(p => (
   837→            <div key={p.id} className="rounded-xl border bg-card shadow-sm p-4">
   838→              <div className="flex flex-wrap items-start justify-between gap-3 mb-3">
   839→                <div>
   840→                  <div className="flex items-center gap-2">
   841→                    <span className="font-mono text-[11px] text-muted-foreground" dir="ltr">{p.code}</span>
   842→                    <StatusPill label={typeLabels[p.type]} />
   843→                  </div>
   844→                  <div className="text-sm font-bold mt-1">{p.title}</div>
   845→                  <div className="text-[11px] text-muted-foreground mt-0.5">
   846→                    ایجاد: {p.createdBy} · {jdate(p.createdAt)} {p.relatedRef && `· مرجع: ${p.relatedRef}`}
   847→                  </div>
   848→                </div>
   849→                <div className="flex items-center gap-1.5">
   850→                  <Button size="sm" variant="ghost" className="h-7 text-xs" onClick={() => setDetailId(p.id)}>جزئیات و لاگ</Button>
   851→                  {p.currentStageIndex < p.stages.length - 1 && (
   852→                    <>
   853→                      <Button size="sm" variant="outline" className="h-7 text-xs gap-1" onClick={() => advance(p)}>تکمیل مرحله <ArrowLeft className="h-3 w-3" /></Button>
   854→                      <Button size="sm" variant="ghost" className="h-7 text-xs text-red-600" onClick={() => rejectStage(p)}><CircleX className="h-3.5 w-3.5" /></Button>
   855→                    </>
   856→                  )}
   857→                </div>
   858→              </div>
   859→              {/* استپر مراحل */}
   860→              <div className="flex items-start overflow-x-auto gap-1 pb-1">
   861→                {p.stages.map((s, i) => (
   862→                  <div key={i} className="flex items-center gap-1 shrink-0">
   863→                    <div className={`rounded-lg border px-2.5 py-1.5 min-w-24 text-center ${i === p.currentStageIndex ? 'ring-2 ring-amber-300' : ''}`}>
   864→                      <div className="text-[10px] text-muted-foreground mb-0.5">{faNumber(i + 1)}</div>
   865→                      <div className="text-[11px] font-medium whitespace-nowrap">{s.name}</div>
   866→                      <div className="mt-1"><StatusPill label={s.responsible} tone={stageTone[s.status]} /></div>
   867→                    </div>
   868→                    {i < p.stages.length - 1 && <div className="h-px w-4 bg-border mt-6" />}
   869→                  </div>
   870→                ))}
   871→              </div>
   872→            </div>
   873→          ))}
   874→        </div>
   875→      )}
   876→
   877→      {/* فرآیند جدید */}
   878→      <FormDialog open={open} onOpenChange={setOpen} title="ایجاد فرآیند جدید" description="قالب چهارمرحله‌ای: شروع → تأیید → اجرا → کنترل">
   879→        <div className="space-y-3">
   880→          <div className="space-y-1.5"><Label>عنوان فرآیند *</Label><Input value={form.title} onChange={e => setForm({ ...form, title: e.target.value })} placeholder="مثلاً: فرآیند آماده‌سازی خودرو برای فروش" /></div>
   881→          <div className="space-y-1.5">
   882→            <Label>نوع</Label>
   883→            <Select value={form.type} onValueChange={v => setForm({ ...form, type: v })}>
   884→              <SelectTrigger><SelectValue /></SelectTrigger>
   885→              <SelectContent>
   886→                {Object.entries(typeLabels).map(([k, v]) => <SelectItem key={k} value={k}>{v}</SelectItem>)}
   887→              </SelectContent>
   888→            </Select>
   889→          </div>
   890→          <div className="space-y-1.5"><Label>مسئول مرحله اول *</Label><Input value={form.responsible} onChange={e => setForm({ ...form, responsible: e.target.value })} /></div>
   891→        </div>
   892→        <div className="flex justify-end gap-2 mt-4">
   893→          <Button variant="outline" onClick={() => setOpen(false)}>انصراف</Button>
   894→          <Button onClick={handleAdd}>ایجاد فرآیند</Button>
   895→        </div>
   896→      </FormDialog>
   897→
   898→      {/* جزئیات و لاگ */}
   899→      <DetailDrawer open={!!detail} onOpenChange={v => !v && setDetailId(null)} title={detail ? `${detail.code} — ${detail.title}` : ''} description={detail ? `مرحله ${faNumber(detail.currentStageIndex + 1)} از ${faNumber(detail.stages.length)}` : ''}>
   900→        {detail && (
   901→          <div className="space-y-4">
   902→            <div className="space-y-2">
   903→              {detail.stages.map((s, i) => (
   904→                <div key={i} className={`rounded-xl border p-3 ${i === detail.currentStageIndex ? 'border-amber-300 bg-amber-50/40 dark:border-amber-500/40 dark:bg-amber-500/10' : ''}`}>
   905→                  <div className="flex items-center justify-between gap-2">
   906→                    <div className="text-sm font-bold">{faNumber(i + 1)}. {s.name}</div>
   907→                    <StatusPill label={s.status === 'done' ? 'تکمیل' : s.status === 'in_progress' ? 'در جریان' : s.status === 'rejected' ? 'رد شده' : 'در انتظار'} tone={stageTone[s.status]} />
   908→                  </div>
   909→                  <div className="text-[11px] text-muted-foreground mt-1">
   910→                    مسئول: {s.responsible}
   911→                    {s.startedAt && ` · شروع: ${jdatetime(s.startedAt)}`}
   912→                    {s.completedAt && ` · پایان: ${jdatetime(s.completedAt)}`}
   913→                  </div>
   914→                  {s.logs.length > 0 && (
   915→                    <div className="mt-2 space-y-1 border-t pt-2">
   916→                      {s.logs.map((l, j) => (
   917→                        <div key={j} className="text-[11px] text-muted-foreground">
   918→                          <b className="text-foreground">{l.action}</b> — {l.by} · {jdatetime(l.at)}{l.note ? ` — ${l.note}` : ''}
   919→                        </div>
   920→                      ))}
   921→                    </div>
   922→                  )}
   923→                </div>
   924→              ))}
   925→            </div>
   926→            <div className="flex gap-2">
   927→              <Input value={logNote} onChange={e => setLogNote(e.target.value)} placeholder="یادداشت برای مرحله جاری..." className="flex-1" />
   928→              <Button onClick={() => addLog(detail)}>ثبت در لاگ</Button>
   929→            </div>
   930→          </div>
   931→        )}
   932→      </DetailDrawer>
   933→    </div>
   934→  );
   935→}
